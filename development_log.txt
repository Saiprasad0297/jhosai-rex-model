## File 5: `src/services/pricing.js`

```js
// ============================================================
// Premium fare engine.
//
//   total = max(minFare, (base + km·perKm + min·perMin) × surge × tier)
//           + bookingFee
// ============================================================

export const FARE_CONFIG = Object.freeze({
  currency: 'USD',
  baseFare: 3.5,       // flag-down charge
  perKm: 1.85,         // distance rate
  perMinute: 0.45,     // time rate (compensates drivers in traffic)
  bookingFee: 2.0,     // flat platform fee
  minimumFare: 12.0,   // floor for short hops
  tiers: Object.freeze({
    premium:   1.0,    // standard premium sedan
    executive: 1.35,   // luxury sedan (E-Class / 5-Series)
    black:     1.75,   // flagship (S-Class / 7-Series)
  }),
});

const round2 = (n) => Math.round(n * 100) / 100;

/**
 * @param {number} distanceKm      Trip distance in kilometers (> 0)
 * @param {number} durationMinutes Trip duration in minutes (> 0)
 * @param {string} [tier]          'premium' | 'executive' | 'black'
 * @param {number} [surgeMultiplier] 1.0 – 5.0 (demand pricing, default 1)
 * @returns Full-fare quote with itemized breakdown
 */
export function calculateFare({ distanceKm, durationMinutes, tier = 'premium', surgeMultiplier = 1.0 }) {
  // --- Validation ---
  const errors = [];
  const dist  = Number(distanceKm);
  const mins  = Number(durationMinutes);
  const surge = Number(surgeMultiplier);

  if (!Number.isFinite(dist) || dist <= 0)              errors.push('"distanceKm" must be a positive number');
  if (!Number.isFinite(mins) || mins <= 0)              errors.push('"durationMinutes" must be a positive number');
  if (!Number.isFinite(surge) || surge < 1 || surge > 5) errors.push('"surgeMultiplier" must be between 1 and 5');
  if (!Object.prototype.hasOwnProperty.call(FARE_CONFIG.tiers, tier)) {
    errors.push(`"tier" must be one of: ${Object.keys(FARE_CONFIG.tiers).join(', ')}`);
  }
  if (errors.length) {
    const err = new Error(errors.join('; '));
    err.statusCode = 400;
    throw err;
  }

  // --- Core math ---
  const distanceCharge = dist * FARE_CONFIG.perKm;
  const timeCharge     = mins * FARE_CONFIG.perMinute;
  const subtotal       = FARE_CONFIG.baseFare + distanceCharge + timeCharge;

  const tierMultiplier = FARE_CONFIG.tiers[tier];
  const surged         = subtotal * surge;
  const tiered         = surged * tierMultiplier;
  const tripFare       = Math.max(tiered, FARE_CONFIG.minimumFare);
  const total          = round2(tripFare + FARE_CONFIG.bookingFee);

  return {
    currency: FARE_CONFIG.currency,
    tier,
    inputs: { distanceKm: dist, durationMinutes: mins, surgeMultiplier: surge },
    breakdown: {
      baseFare: round2(FARE_CONFIG.baseFare),
      distanceCharge: round2(distanceCharge),
      timeCharge: round2(timeCharge),
      subtotal: round2(subtotal),
      surgeMultiplier: surge,
      surgeAmount: round2(surged - subtotal),
      tierMultiplier,
      tierAdjustment: round2(tiered - surged),
      minimumFareApplied: tiered < FARE_CONFIG.minimumFare,
      tripFare: round2(tripFare),
      bookingFee: FARE_CONFIG.bookingFee,
    },
    total,
  };
}
```

## File 6: `src/routes/drivers.js`

```js
import { Router } from 'express';
import { drivers, DRIVER_STATUSES, findDriverById } from '../data/mockDb.js';

const router = Router();
const VALID_STATUSES = Object.values(DRIVER_STATUSES);

/**
 * GET /api/drivers            — list all drivers
 * GET /api/drivers?status=…   — filter: available | offline | on-trip
 */
router.get('/', (req, res) => {
  const { status } = req.query;

  if (status) {
    if (!VALID_STATUSES.includes(status)) {
      return res.status(400).json({
        error: `Invalid status "${status}". Valid values: ${VALID_STATUSES.join(', ')}`,
      });
    }
    const filtered = drivers.filter((d) => d.status === status);
    return res.json({ count: filtered.length, drivers: filtered });
  }

  res.json({ count: drivers.length, drivers });
});

/** GET /api/drivers/available — convenience shortcut */
router.get('/available', (_req, res) => {
  const available = drivers.filter((d) => d.status === DRIVER_STATUSES.AVAILABLE);
  res.json({ count: available.length, drivers: available });
});

/** GET /api/drivers/:id */
router.get('/:id', (req, res) => {
  const driver = findDriverById(req.params.id.toUpperCase());
  if (!driver) return res.status(404).json({ error: `Driver ${req.params.id} not found` });
  res.json({ driver });
});

/**
 * PATCH /api/drivers/:id/status
 * Body: { "status": "available" | "offline" | "on-trip" }
 */
router.patch('/:id/status', (req, res) => {
  const { status } = req.body ?? {};

  if (!VALID_STATUSES.includes(status)) {
    return res.status(400).json({
      error: `Body must include a valid "status". Valid values: ${VALID_STATUSES.join(', ')}`,
    });
  }

  const driver = findDriverById(req.params.id.toUpperCase());
  if (!driver) return res.status(404).json({ error: `Driver ${req.params.id} not found` });

  driver.status = status;
  res.json({ message: `Driver ${driver.id} is now "${status}"`, driver });
});

export default router;
```

```js
import { Router } from 'express';
import { riders, nextRiderId, findRiderById } from '../data/mockDb.js';

const router = Router();
const EMAIL_RE = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;

/** GET /api/riders */
router.get('/', (_req, res) => res.json({ count: riders.length, riders }));

/** GET /api/riders/:id */
router.get('/:id', (req, res) => {

  res.json({ rider });
});

/**
 * POST /api/riders — register a rider
 * Body: { name, email, phone? }
 */
router.post('/', (req, res) => {
  const name  = typeof req.body?.name  === 'string' ? req.body.name.trim() : '';
  const email = typeof req.body?.email === 'string' ? req.body.email.trim().toLowerCase() : '';
  const phone = typeof req.body?.phone === 'string' ? req.body.phone.trim() : '';

  const errors = [];
  if (name.length < 2) errors.push('"name" is required (min 2 characters)');
  if (!EMAIL_RE.test(email)) errors.push('"email" must be a valid email address');
  if (errors.length) return res.status(400).json({ errors });

  if (riders.some((r) => r.email === email)) {
    return res.status(409).json({ error: 'A rider with this email already exists' });
  }

  const rider = {
    id: nextRiderId(),
    name,
    email,
    phone: phone || null,
    createdAt: new Date().toISOString(),
  };
  riders.push(rider);
  res.status(201).json({ message: 'Rider registered', rider });
});

export default router;
```

```js
import { Router } from 'express';
import { calculateFare } from '../services/pricing.js';
import {
  rides, nextRideId, findRiderById, findAvailableDrivers, DRIVER_STATUSES,
} from '../data/mockDb.js';

const router = Router();

/**
 * POST /api/rides/estimate
 * Body: { distanceKm, durationMinutes, tier?, surgeMultiplier? }
 */
router.post('/estimate', (req, res) => {
  try {
    res.json({ estimate: calculateFare(req.body ?? {}) });
  } catch (err) {
    res.status(err.statusCode || 500).json({ error: err.message });
  }
});

/**
 * POST /api/rides — book a ride (assigns the first available driver)
 * Body: { riderId, distanceKm, durationMinutes, tier?, surgeMultiplier? }
 */
router.post('/', (req, res) => {
  try {
    const { riderId, distanceKm, durationMinutes, tier, surgeMultiplier } = req.body ?? {};

    const rider = riderId ? findRiderById(String(riderId).toUpperCase()) : null;
    if (!rider) return res.status(404).json({ error: 'Rider not found — provide a valid "riderId"' });

    const quote = calculateFare({ distanceKm, durationMinutes, tier, surgeMultiplier });

    const [driver] = findAvailableDrivers();
    if (!driver) return res.status(503).json({ error: 'No drivers available right now. Try again shortly.' });

    // Status transition: available → on-trip
    driver.status = DRIVER_STATUSES.ON_TRIP;

    const ride = {
      id: nextRideId(),
      riderId: rider.id,
      driverId: driver.id,
      tier: quote.tier,
      fare: quote.total,
      currency: quote.currency,
      status: 'in-progress',
      requestedAt: new Date().toISOString(),
    };
    rides.push(ride);

    res.status(201).json({ ride, driver, estimate: quote });
  } catch (err) {
    res.status(err.statusCode || 500).json({ error: err.message });
  }
});

/** GET /api/rides — ride history (mock) */
router.get('/', (_req, res) => res.json({ count: rides.length, rides }));

export default router;
```
